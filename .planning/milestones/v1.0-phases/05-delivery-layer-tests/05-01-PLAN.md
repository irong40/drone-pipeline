--- collect_photos ---
test_collect_photos_returns_sorted_jpegs(tmp_path):
  jpeg_dir = tmp_path / "photos" / "jpeg"
  jpeg_dir.mkdir(parents=True)
  (jpeg_dir / "DJI_0002.jpg").write_bytes(b"")
  (jpeg_dir / "DJI_0001.JPG").write_bytes(b"")
  (jpeg_dir / "thumb.PNG").write_bytes(b"")   # excluded
  result = collect_photos(str(tmp_path))
  assert len(result) == 2
  assert all(p.endswith((".jpg", ".JPG")) for p in result)
  assert "DJI_0001" in result[0]

test_collect_photos_missing_dir_returns_empty(tmp_path):
  assert collect_photos(str(tmp_path)) == []
