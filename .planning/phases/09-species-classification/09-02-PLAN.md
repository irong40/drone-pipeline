```python
def estimate_cost(canopy_count, skip_plantnet):
    """Estimate API cost based on canopy count."""
    openai_cost_per = 0.02  # ~$0.02 per gpt-4o vision call (low-res)
    plantnet_cost_per = 0.00  # Free tier
    estimated = canopy_count * openai_cost_per
    if not skip_plantnet:
        estimated += canopy_count * 0.001  # minimal overhead
    return round(estimated, 2)
```

```python
def update_supabase(client, results):
    """Batch update vegetation_detections with species data."""
    for detection_id, final in results:
        client.table("vegetation_detections").update({
            "species_tag": final["species_tag"],
            "species_confidence": final["species_confidence"],
            "vegetation_type": final["vegetation_type"],
            "cross_validated": final["cross_validated"],
            "classification_details": final["classification_details"],
```
